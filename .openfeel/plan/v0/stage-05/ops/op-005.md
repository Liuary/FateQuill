## 实施步骤
- [ ] 1. `useSettingCards.ts`：按作品加载 + CRUD（复用仓储 + reload）
  ```ts
  import { useCallback, useEffect, useState } from "react";
  import type { SettingCard } from "@/domain/models/setting-card";
  import { repositories } from "@/ipc/repositories";

  /** 某作品的设定卡列表 + 增删改（持久化经 stage-02 仓储） */
  export function useSettingCards(novelId: number | null) {
    const [cards, setCards] = useState<SettingCard[]>([]);
    const reload = useCallback(async () => { setCards(novelId == null ? [] : await repositories.settingCard.listByNovel(novelId)); }, [novelId]);
    useEffect(() => { void reload(); }, [reload]);
    const create = useCallback((input: { title: string; content: string; kind: string }) => { if (novelId==null) return Promise.resolve(); return repositories.settingCard.create({ novelId, ...input }).then(reload); }, [novelId, reload]);
    const update = useCallback((id: number, input: { title: string; content: string; kind: string }) => repositories.settingCard.update(id, input).then(reload), [reload]);
    const remove = useCallback((id: number) => repositories.settingCard.remove(id).then(reload), [reload]);
    return { cards, reload, create, update, remove };
  }
  ```
