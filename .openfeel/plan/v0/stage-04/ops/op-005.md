- [ ] 1. `src/features/editor/useOutline.ts`：加载/操作卷章（复用 `@/ipc/repositories`）
  ```ts
  import { useCallback, useEffect, useState } from "react";
  import type { Volume } from "@/domain/models/volume";
  import type { Chapter } from "@/domain/models/chapter";
  import { repositories } from "@/ipc/repositories";

  /** 加载某作品的卷+章（章按卷分组）；提供结构操作并刷新 */
  export function useOutline(novelId: number | null) {
    const [volumes, setVolumes] = useState<Volume[]>([]);
    const [chaptersByVolume, setChaptersByVolume] = useState<Record<number, Chapter[]>>({});

    const reload = useCallback(async () => {
      if (novelId == null) { setVolumes([]); setChaptersByVolume({}); return; }
      const vols = (await repositories.volume.listByNovel(novelId)).sort((a, b) => a.orderIndex - b.orderIndex);
      const map: Record<number, Chapter[]> = {};
      for (const v of vols) {
        map[v.id] = (await repositories.chapter.listByVolume(v.id)).sort((a, b) => a.orderIndex - b.orderIndex);
      }
      setVolumes(vols); setChaptersByVolume(map);
    }, [novelId]);

    useEffect(() => { void reload(); }, [reload]);
    return { volumes, chaptersByVolume, reload };
  }
  ```
