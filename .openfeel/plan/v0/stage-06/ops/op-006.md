- [x] 1. `useReview.ts`（编排：审查/改判/重写/采纳）
  ```ts
  import type { Editor } from "@tiptap/react";
  import { createEditorController } from "@/features/editor/EditorController";
  import { runReviewLoop } from "@/orchestration/review/loop";
  import { useReviewStore } from "@/store/reviewStore";

  /** 审查编排：运行四维评审回路 → 入池；改判；触发重写；采纳（replaceContent） */
  export function useReview(editor: Editor | null, deps: { evaluators; provider; model; context; content: string }) {
    const runReview = async () => {
      const { needsHuman } = await runReviewLoop({
        provider: deps.provider, model: deps.model, content: deps.content, context: deps.context,
        evaluators: deps.evaluators, weights: useReviewStore.getState().weights,
        autoRewrite: useReviewStore.getState().autoRewrite, maxRounds: useReviewStore.getState().maxRounds,
        onVersion: (v) => useReviewStore.getState().addVersion(v),   // 入池，不替换
      });
      useReviewStore.getState().setNeedsHumanReview(needsHuman);
    };
    const rejudge = (versionId: string, dimension: ReviewDimension, score: number) => { /* 覆盖分 + 重算总分 */ };
    const triggerRewrite = async () => { /* 用户显式触发一次重写回路（无视 autoRewrite 开关） */ };
    const adopt = async (versionId: string) => {
      const v = useReviewStore.getState().versions.find((x) => x.id === versionId);
      if (!editor || !v) return;
      createEditorController(editor).replaceContent(v.content);   // 单条撤销
      useReviewStore.getState().setActive(versionId);
    };
    return { runReview, rejudge, triggerRewrite, adopt };
  }
  ```
