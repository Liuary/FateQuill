## 目标
① **REV-018**：为交叉判断（T2）补 **UI 触发入口 + 编排 hook**——研究台新增「开始交叉判断」（对已有采样候选逐模型摘取 → 精确交集合并 → 入 `pendingResults` 待确认队列），使 **`multi_model_cross` 通道可达**（修复 DoD「三采集通道」断裂）；② **REV-019**：`useAnnotation.save` 捕获 IPC 失败并提示（与 `useSkillLibrary` 模式一致），消除静默失败；③ REV-020 杂项顺带（CSV ISO / seq 说明 / manual 口径）。

## 缺陷根因（取证）
- **REV-018**：`extractFlavorExcerpts`/`mergeByExcerpt`/`addPendingResults` **生产代码零调用**；`ResearchWorkbench.tsx` 仅采样入口，无交叉判断按钮/hook → `pendingResults` 恒空 → `multi_model_cross` 通道不可达。`cross-judge.test.ts` 仅覆盖纯函数（测试通过 ≠ 功能可用）。
- **REV-019**：`useAnnotation.save` 仅 `finally` 无 `catch`；`AnnotationPanel.handleSave` `await save(...)` 未捕获 → unhandled rejection、失败无提示（`savedCount` 不变但用户无感）。

- [x] 2. 新增 `src/features/research/useCrossJudge.ts`（**REV-018 核心**）
  ```ts
  /** 交叉判断编排：对已有采样候选，逐模型摘取 → 精确交集合并 → 入待确认队列 */
  export function useCrossJudge(models: SamplingModel[]) {
    const [crossing, setCrossing] = useState(false);
    const abortRef = useRef<AbortController | null>(null);
    const runCrossJudge = useCallback(async () => {
      const candidates = useResearchStore.getState().candidates;
      if (candidates.length === 0 || models.length === 0) return;   // 无候选/无模型：不启动
      const controller = new AbortController(); abortRef.current = controller; setCrossing(true);
      const merged: CrossJudgeResult[] = [];
      try {
        for (const candidate of candidates) {
          if (controller.signal.aborted) break;
          const perModel: ModelExcerpts[] = [];
          for (const m of models) {
            if (controller.signal.aborted) break;
            try {
              perModel.push(await extractFlavorExcerpts({ provider: m.provider, model: m.model, content: candidate.content, temperature: 0 }));
            } catch { /* REV-018：单模型摘取失败**容错跳过**，不整体失败 */ }
          }
          if (perModel.length > 0) merged.push(...mergeByExcerpt(perModel));
        }
        if (merged.length > 0) useResearchStore.getState().addPendingResults(merged); // 入待确认队列（sourceType=multi_model_cross）
      } finally { abortRef.current = null; setCrossing(false); }
    }, [models]);
    const stopCrossJudge = useCallback(() => { abortRef.current?.abort(); }, []);
    return { runCrossJudge, stopCrossJudge, crossing };
  }
  ```
  - 复用 `extractFlavorExcerpts`/`mergeByExcerpt`（`@/orchestration/research/cross-judge`）；结果 `sourceType="multi_model_cross"`（由 `mergeByExcerpt` 携带）。
