## 目标
建立三段式落点：`src/domain/repositories/`（纯 TS 仓储接口）→ `src/ipc/repositories/`（经 `invoke` 实现接口）→ `src-tauri/`（op-004 命令）；Vitest（mock `invoke`）验证命令名/参数/返回映射与错误路径。**不新建 `src/infra/`**。

## 分层约定（plan v2，REV-003）
- 接口层纯 TS，仅依赖 `@/domain/models/*`（**不得**依赖 `@/ipc`，受 op-002 ESLint 纯度规则约束）。
- 实现层 `src/ipc/repositories/` 实现接口，经 `@/ipc/client` 的 `invokeCommand` 调 Rust 命令；负责 snake_case 行 ↔ camelCase 领域模型的映射。
- 组件/feature 只依赖接口（`NovelRepository` 等），不感知命令名。

- [ ] 3. 实现 `src/ipc/repositories/novel-repository.ts`
  ```ts
  import { invokeCommand } from "@/ipc/client";
  import type { Novel } from "@/domain/models/novel";
  import type { NovelRepository } from "@/domain/repositories/novel-repository";

  interface NovelRow { id: number; title: string; synopsis: string; created_at: string; updated_at: string }
  const toNovel = (r: NovelRow): Novel => ({ id: r.id, title: r.title, synopsis: r.synopsis, createdAt: r.created_at, updatedAt: r.updated_at });

  export function createNovelRepository(): NovelRepository {
    return {
      async list() { return (await invokeCommand<NovelRow[]>("list_novels")).map(toNovel); },
      async get(id) { return toNovel(await invokeCommand<NovelRow>("get_novel", { id })); },
      async create(input) { return toNovel(await invokeCommand<NovelRow>("create_novel", { title: input.title, synopsis: input.synopsis })); },
      async update(id, input) { return toNovel(await invokeCommand<NovelRow>("update_novel", { id, title: input.title, synopsis: input.synopsis })); },
      async remove(id) { await invokeCommand<void>("delete_novel", { id }); },
    };
  }
  ```
  - 其余 4 个仓储同构实现（volume/chapter/setting-card/character），命令名与 op-004 一致；`setting-card`/`character` 的 `get(id)` 分别映射 `get_setting_card`/`get_character`（不得缺省）。`chapter` 的行映射含 `content_format`→`contentFormat`、`word_count`→`wordCount`、`order_index`→`orderIndex`、`volume_id`→`volumeId`、`status`。

- [ ] 5. Vitest（mock `@tauri-apps/api/core`）覆盖命令名/参数/映射/错误路径
  - `src/ipc/repositories/novel-repository.test.ts`（示例）：
    ```ts
    import { beforeEach, describe, expect, it, vi } from "vitest";
    import { invoke } from "@tauri-apps/api/core";
    import { createNovelRepository } from "./novel-repository";
    import { IpcError } from "@/ipc/errors";

    vi.mock("@tauri-apps/api/core", () => ({ invoke: vi.fn() }));

    describe("novel repository", () => {
      beforeEach(() => vi.mocked(invoke).mockReset());
      it("list 映射 snake_case→camelCase", async () => {
        vi.mocked(invoke).mockResolvedValue([{ id: 1, title: "t", synopsis: "s", created_at: "c", updated_at: "u" }]);
        const out = await createNovelRepository().list();
        expect(invoke).toHaveBeenCalledWith("list_novels", undefined);
        expect(out[0]).toEqual({ id: 1, title: "t", synopsis: "s", createdAt: "c", updatedAt: "u" });
      });
      it("create 透传 camelCase 参数", async () => {
        vi.mocked(invoke).mockResolvedValue({ id: 2, title: "x", synopsis: "", created_at: "c", updated_at: "u" });
        await createNovelRepository().create({ title: "x", synopsis: "" });
        expect(invoke).toHaveBeenCalledWith("create_novel", { title: "x", synopsis: "" });
      });
      it("错误路径归一化为 IpcError", async () => {
        vi.mocked(invoke).mockRejectedValue({ code: "NOT_FOUND", message: "novel not found" });
        await expect(createNovelRepository().get(9)).rejects.toBeInstanceOf(IpcError);
      });
    });
    ```
