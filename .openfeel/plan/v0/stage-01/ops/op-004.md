- 新增：`eslint.config.js`、`.prettierrc`、`.prettierignore`、`vitest.config.ts`、`src/test/setup.ts`、`src/lib/utils.test.ts`
- 修改：`package.json`（scripts）、`src-tauri/src/lib.rs`（新增 `#[cfg(test)]`）
- 生成的验证产物：`.eslintcache`（仅当 ESLint 使用 `--cache` 时；本 op 默认 `eslint .` 不生成，且须确保其被 `.gitignore` 忽略——见步骤 11）

## 实施步骤
- [ ] 1. 安装 ESLint 依赖
  ```powershell
  corepack pnpm add -D eslint@10.12.0 @eslint/js@10.0.1 typescript-eslint@8.71.1 eslint-plugin-react-hooks@7.1.1 eslint-plugin-react-refresh@0.5.7 globals@17.13.0
  ```
- [ ] 2. 新建 `eslint.config.js`（flat config）
  ```js
  import js from "@eslint/js";
  import globals from "globals";
  import reactHooks from "eslint-plugin-react-hooks";
  import reactRefresh from "eslint-plugin-react-refresh";
  import tseslint from "typescript-eslint";

  export default tseslint.config(
    { ignores: ["dist", "node_modules", "src-tauri/target", "src-tauri/gen"] },
    {
      files: ["**/*.{ts,tsx}"],
      extends: [js.configs.recommended, ...tseslint.configs.recommended],
      languageOptions: { ecmaVersion: 2022, globals: globals.browser },
      plugins: { "react-hooks": reactHooks, "react-refresh": reactRefresh },
      rules: {
        ...reactHooks.configs.recommended.rules,
        "react-refresh/only-export-components": ["warn", { allowConstantExport: true }],
      },
    },
  );
  ```
- [ ] 3. 安装 Prettier 依赖并配置
  ```powershell
  corepack pnpm add -D prettier@3.9.9 prettier-plugin-tailwindcss@0.8.1
  ```
  新建 `.prettierrc`：
  ```json
  { "semi": true, "singleQuote": false, "trailingComma": "all", "printWidth": 100, "plugins": ["prettier-plugin-tailwindcss"] }
  ```
  新建 `.prettierignore`：`dist`、`node_modules`、`pnpm-lock.yaml`、`src-tauri/target`、`src-tauri/gen`、`.openfeel`。
- [ ] 4. 安装 Vitest 依赖
  ```powershell
  corepack pnpm add -D vitest@5.0.3 @testing-library/react@16.3.3 @testing-library/jest-dom@7.0.1 @testing-library/user-event@14.6.7 jsdom@30.1.2
  ```
- [ ] 5. 新建 `vitest.config.ts`（复用 vite 的 `@` 别名）
  ```ts
  import { defineConfig, mergeConfig } from "vitest/config";
  import viteConfig from "./vite.config";

  export default mergeConfig(
    viteConfig({ mode: "test", command: "serve" }),
    defineConfig({
      test: {
        environment: "jsdom",
        globals: true,
        setupFiles: ["./src/test/setup.ts"],
        include: ["src/**/*.{test,spec}.{ts,tsx}"],
      },
    }),
  );
  ```
- [ ] 6. 新建 `src/test/setup.ts`
  ```ts
  import "@testing-library/jest-dom/vitest";
  ```
- [ ] 7. 新建示例测试 `src/lib/utils.test.ts`（稳定，不随 UI 变动）
  ```ts
  import { describe, expect, it } from "vitest";
  import { cn } from "@/lib/utils";

  describe("cn", () => {
    it("合并类名", () => {
      expect(cn("a", "b")).toBe("a b");
    });
    it("处理 Tailwind 冲突", () => {
      expect(cn("p-2", "p-4")).toBe("p-4");
    });
  });
  ```
- [ ] 8. `package.json` 增加 scripts（保留既有 `dev`/`build`/`preview`/`tauri`）
  ```json
  "lint": "eslint .",
  "lint:fix": "eslint . --fix",
  "format": "prettier --write .",
  "format:check": "prettier --check .",
  "test": "vitest run",
  "test:watch": "vitest",
  "test:rust": "cargo test --manifest-path src-tauri/Cargo.toml"
  ```
- [ ] 9. 在 `src-tauri/src/lib.rs` 末尾新增 Rust 单元测试（示例，至少 1 个）
  ```rust
  #[cfg(test)]
  mod tests {
      use super::*;

      #[test]
      fn sample_test_runs() {
          assert_eq!(2 + 2, 4);
      }
  }
  ```
  > 注意：op-005 会将 `greet` 替换为 `ping`，届时同步更新本测试（无需依赖 `greet`）。
- [ ] 10. 运行门禁
  ```powershell
  corepack pnpm format        # 首次统一格式化（避免 format:check 误报）
  corepack pnpm lint
  corepack pnpm test
  corepack pnpm format:check
  & "$env:USERPROFILE\.cargo\bin\cargo.exe" test --manifest-path src-tauri/Cargo.toml
  ```
  > `format:check` 属可选门禁（CI 不含格式化检查）；若仍报未格式化文件，执行 `corepack pnpm format` 后再检查。
- [ ] 11. 忽略缓存并**提交本 op 变更（REV-009 / REV-014①）**
  ```powershell
  # 确保 ESLint 缓存文件不入库（op-007 将合并进最终 .gitignore）
  if (-not (Select-String -Path .gitignore -Pattern '^\.eslintcache$' -Quiet)) { Add-Content -Path .gitignore -Value ".eslintcache" }
  git add -A
  git commit -m "chore(tooling): add ESLint, Prettier, Vitest and cargo test setup (T4)"
  ```
